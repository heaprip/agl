# Нетто-стехиометрия одного пути.
rule path_net strict @: path_net(pathway: ?p, species: ?x, value: ?v) <-
    pathway(id: ?p), top(species: ?x),
    ?v = sum(?d * ?t * ?sv * ?c) over (
        step(pathway: ?p, reaction: ?r, direction: ?dir, times: ?t),
        dir_value(direction: ?dir, value: ?d),
        participates(reaction: ?r, species: ?y, side: ?side, coeff: ?c),
        side_value(side: ?side, value: ?sv),
        member(species: ?y, of: ?x))

# Нетто-стехиометрия системы: набор путей с кратностями.
rule sys_net strict @: sys_net(system: ?s, species: ?x, value: ?v) <-
    system(id: ?s), top(species: ?x),
    ?v = sum(?u * ?d * ?t * ?sv * ?c) over (
        uses(system: ?s, pathway: ?p, times: ?u),
        step(pathway: ?p, reaction: ?r, direction: ?dir, times: ?t),
        dir_value(direction: ?dir, value: ?d),
        participates(reaction: ?r, species: ?y, side: ?side, coeff: ?c),
        side_value(side: ?side, value: ?sv),
        member(species: ?y, of: ?x))

rule consumed strict @: consumed(system: ?s, species: ?x) <-
    sys_net(system: ?s, species: ?x, value: ?v), ?v < 0

# Вещество потребляется, но системой не производится и в среде отсутствует.
rule lacking strict @: lacking(system: ?s, species: ?x) <-
    consumed(system: ?s, species: ?x),
    ?k = count(?x) over (available(species: ?x)), ?k = 0

rule supplied strict @: supplied(system: ?s, species: ?x) <-
    consumed(system: ?s, species: ?x), available(species: ?x)

rule viable strict @: viable(system: ?s) <-
    system(id: ?s),
    ?c = count(?x) over (consumed(system: ?s, species: ?x)),
    ?a = count(?x) over (supplied(system: ?s, species: ?x)),
    ?c = ?a

rule not_viable strict @: ¬viable(system: ?s) <- lacking(system: ?s, species: ?x)
