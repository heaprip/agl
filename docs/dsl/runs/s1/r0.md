# Пируватдекарбоксилаза: пируват + H+ -> ацетальдегид + CO2.

fact pdc_co2: participates(reaction: pdc, species: CHEBI_16526, side: right, coeff: 1) @ [llm_proposal]

# Алкогольдегидрогеназа: ацетальдегид + NADH + H+ -> этанол + NAD+.
fact rx_adh: reaction(id: adh_ethanol) @ [llm_proposal]
fact adh_aa: participates(reaction: adh_ethanol, species: CHEBI_15343, side: left, coeff: 1) @ [llm_proposal]
fact adh_nadh: participates(reaction: adh_ethanol, species: CHEBI_57945, side: left, coeff: 1) @ [llm_proposal]

# Дыхательная цепь, укрупнённо: 2 NADH + 2 H+ + O2 -> 2 NAD+ + 2 H2O.

fact glycolysis_atp: glycolysis_atp(value: 2) @ [llm_proposal]

# Стехиометрия системы.
rule sys_net strict @: sys_net(system: ?s, species: ?x, value: ?v) <-
    system(id: ?s), top(species: ?x),
    ?v = sum(?u * ?d * ?t * ?sv * ?c) over (uses(system: ?s, pathway: ?p, times: ?u),
                                           member(species: ?y, of: ?x),
                                           step(pathway: ?p, reaction: ?r, direction: ?dir, times: ?t),
                                           dir_value(direction: ?dir, value: ?d),
                                           participates(reaction: ?r, species: ?y, side: ?side, coeff: ?c),
                                           side_value(side: ?side, value: ?sv))

rule atp_yield strict @: atp_yield(system: ?s, value: ?v) <-
    sys_net(system: ?s, species: CHEBI_30616, value: ?v)

rule no_extra_atp strict @: yields_atp_beyond_glycolysis(kind: ?k, value: ?z) <-
    fermentation_system(kind: ?k, system: ?s),
    atp_yield(system: ?s, value: ?v),
    glycolysis_atp(value: ?g),
    ?z = ?v - ?g

rule regen_nad strict @ anaerobic: regenerates_nad(kind: ?k) <-
    fermentation_system(kind: ?k, system: ?s),
    sys_net(system: ?s, species: CHEBI_57540, value: 0),
    sys_net(system: ?s, species: CHEBI_57945, value: 0)

query sys_net(system: alcohol_fermentation_system, species: CHEBI_57945, value: 0) @ anaerobic expect yes

query yields_atp_beyond_glycolysis(kind: lactate_fermentation, value: 0) @ expect yes
query yields_atp_beyond_glycolysis(kind: alcohol_fermentation, value: 0) @ expect yes

query reoxidizes_nadh(species: CHEBI_57945, by: respiratory_chain) @ anaerobic expect unknown
